# osctrl Modern Frontend — v1 Nodes Page PoC Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a SvelteKit-based Nodes table page mounted at `/ui/nodes`, embedded into the `osctrl-admin` Go binary, served behind the existing session-cookie auth, side-by-side with the legacy template UI. Spec: `docs/superpowers/specs/2026-05-10-nodes-poc-design.md`.

**Architecture:** New `cmd/admin/web/` SvelteKit project, output baked into the Go binary via `//go:embed`. Two new route mounts in `main.go` (`GET /ui/`, `GET /ui/`). One tiny new bootstrap JSON handler so the SPA can discover the user's environments. Existing `GET /paginated-json/environment/{env}/{target}` is reused unchanged. Build-tag escape hatch (`dev_no_embed`) lets Go run without forcing a SPA build.

**Tech Stack:** SvelteKit 2 + Svelte 5 (static adapter, SPA mode), TypeScript strict, Tailwind CSS v4, shadcn-svelte, TanStack Query (Svelte) for server state, TanStack Table for headless table logic, Vite (HMR + dev proxy), Vitest + `@testing-library/svelte` + Playwright for tests.

**Verified facts about the existing codebase:**

- Existing nodes pagination endpoint: `GET /paginated-json/environment/{env}/{target}` (registered at `cmd/admin/main.go:423-425`). DataTables-shape via **GET query string**, not POST body. Targets: `all`, `active`, `inactive`. Response: `{ draw, recordsTotal, recordsFiltered, data }`.
- `data[]` element type `NodeJSON` in `cmd/admin/handlers/json-nodes.go:42-54` (fields: `checkbox, uuid, username, localname, ip, platform, version, osquery, lastseen, firstseen` where `lastseen` and `firstseen` are `CreationTimes`).
- Auth wrapper: `handlerAuthCheck(...)` in `cmd/admin/main.go`, used on every authenticated route. Unauthenticated requests redirect to `/login`.
- Login flow: `GET /login` and `POST /login` (legacy HTML), session cookie set by `LoginPOSTHandler`. Logout: `GET /logout`.
- Static-asset route: `GET /static/` (legacy assets) — **untouched**. New route prefix is `/ui/`.
- ServeMux: stdlib `net/http.ServeMux` (Go 1.22+ pattern syntax with `{wildcard}` and method prefixes).

package handlers

import (
	"encoding/json"
	"net/http"

	"github.com/jmpsec/osctrl/cmd/admin/sessions"
	"github.com/jmpsec/osctrl/pkg/users"
	"github.com/jmpsec/osctrl/pkg/utils"
	"github.com/rs/zerolog/log"
)

// BootstrapResponse is the payload the SPA fetches on first load.
type BootstrapResponse struct {
	User string   `json:"user"`
	Envs []string `json:"envs"`
}

// JSONBootstrapHandler returns the current user and the environments they can access.
// Read-only. Used by the SvelteKit SPA shell to populate the env switcher.
func (h *HandlersAdmin) JSONBootstrapHandler(w http.ResponseWriter, r *http.Request) {
	if h.DebugHTTPConfig.EnableHTTP {
		utils.DebugHTTPDump(h.DebugHTTP, r, h.DebugHTTPConfig.ShowBody)
	}
	ctx := r.Context().Value(sessions.ContextKey(sessions.CtxSession)).(sessions.ContextValue)
	user := ctx[sessions.CtxUser]

	envs := []string{}
	allEnvs, err := h.Envs.All()
	if err != nil {
		log.Err(err).Msg("error listing environments")
		http.Error(w, "internal error", http.StatusInternalServerError)
		return
	}
	for _, e := range allEnvs {
		if h.Users.CheckPermissions(user, users.UserLevel, e.UUID) {
			envs = append(envs, e.Name)
		}
	}

	w.Header().Set("Content-Type", "application/json")
	if err := json.NewEncoder(w).Encode(BootstrapResponse{User: user, Envs: envs}); err != nil {
		log.Err(err).Msg("error encoding bootstrap response")
	}
}
```

//go:build !dev_no_embed

package main

import (
	"embed"
	"io/fs"
	"net/http"
	"strings"
)

//go:embed all:web/build
var webFS embed.FS

// newWebHandler serves the embedded SvelteKit SPA at /ui/*.
// Any request whose path-after-/ui/ is not an existing file in the
// embedded FS falls back to index.html so client-side routes resolve.
func newWebHandler() http.Handler {
	sub, err := fs.Sub(webFS, "web/build")
	if err != nil {
		panic("web/build not found in embed: " + err.Error())
	}
	fileServer := http.FileServer(http.FS(sub))

	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		// Strip /ui prefix so embed FS lookups are relative to web/build.
		// Without this, http.FileServer would look up /ui/foo, but the embed
		// FS only knows about /foo.
		stripped := strings.TrimPrefix(r.URL.Path, "/ui")
		if stripped == "" {
			stripped = "/"
		}

		// Long-cache for hashed assets, no-cache for index.html.
		if strings.HasPrefix(stripped, "/_app/immutable/") {
			w.Header().Set("Cache-Control", "public, max-age=31536000, immutable")
		} else {
			w.Header().Set("Cache-Control", "no-cache")
		}

		// SPA fallback: if the file doesn't exist, serve index.html.
		// We probe the FS first.
		probe := strings.TrimPrefix(stripped, "/")
		if probe != "" && probe != "index.html" {
			if _, err := sub.Open(probe); err != nil {
				stripped = "/"
			}
		}

		r2 := r.Clone(r.Context())
		r2.URL.Path = stripped
		fileServer.ServeHTTP(w, r2)
	})
}
```

import { base } from '$app/paths';

export class AuthError extends Error {
  constructor(msg = 'authentication required') { super(msg); this.name = 'AuthError'; }
}

export class ApiError extends Error {
  constructor(public status: number, msg: string) { super(msg); this.name = 'ApiError'; }
}

export interface FetchOptions extends RequestInit {
  /** Path relative to origin, e.g. '/paginated-json/environment/dev/all'. */
  path: string;
}

export async function apiFetch(opts: FetchOptions): Promise<Response> {
  const { path, ...init } = opts;
  const res = await fetch(path, {
    ...init,
    credentials: 'include',
    headers: { Accept: 'application/json', ...(init.headers || {}) }
  });

  if (res.status === 401 || res.status === 302) {
    // Redirect to legacy login, preserving where we were headed.
    if (typeof window !== 'undefined') {
      const next = encodeURIComponent(window.location.pathname + window.location.search);
      window.location.href = `/login?next=${next}`;
    }
    throw new AuthError();
  }
  if (!res.ok) {
    const body = await res.text().catch(() => '');
    throw new ApiError(res.status, body || res.statusText);
  }
  return res;
}

export async function apiJSON<T>(opts: FetchOptions): Promise<T> {
  const res = await apiFetch(opts);
  return res.json() as Promise<T>;
}

// Used in tests to construct expected URLs without hard-coding base.
export function withBase(p: string): string {
  return base + p;
}

import { apiJSON } from './client';
import type { DataTablesPayload, NodeJSON, NodesPage, NodeTarget } from './types';

export interface SearchNodesParams {
  page: number;            // 0-indexed
  pageSize: number;
  search: string;
  sortColumn?: string;     // column name as expected by mapDTColumnToDB on the Go side
  sortDir?: 'asc' | 'desc';
}

/**
 * Calls GET /paginated-json/environment/{env}/{target} with DataTables-shaped
 * query string and adapts the response to a clean shape.
 *
 * This is the ONE place in the SPA that touches the legacy DataTables format.
 * If the backend payload is ever cleaned up, only this function changes.
 */
export async function searchNodes(
  env: string,
  target: NodeTarget,
  params: SearchNodesParams
): Promise<NodesPage> {
  const qs = new URLSearchParams();
  qs.set('draw', '1');
  qs.set('start', String(params.page * params.pageSize));
  qs.set('length', String(params.pageSize));
  qs.set('search', params.search);
  if (params.sortColumn !== undefined) {
    qs.set('order[0][column]', params.sortColumn);
    qs.set('order[0][dir]', params.sortDir ?? 'asc');
  }
  const path = `/paginated-json/environment/${encodeURIComponent(env)}/${encodeURIComponent(target)}?${qs.toString()}`;
  const payload = await apiJSON<DataTablesPayload<NodeJSON>>({ path, method: 'GET' });
  return {
    rows: payload.data ?? [],
    totalRows: payload.recordsTotal ?? 0,
    filteredRows: payload.recordsFiltered ?? 0
  };
}

const STORAGE_KEY = 'osctrl.selectedEnv';

let _env = $state<string | null>(loadInitial());

function loadInitial(): string | null {
  if (typeof localStorage === 'undefined') return null;
  return localStorage.getItem(STORAGE_KEY);
}

export function getSelectedEnv(): string | null {
  return _env;
}

export function setSelectedEnv(name: string | null) {
  _env = name;
  if (typeof localStorage !== 'undefined') {
    if (name === null) localStorage.removeItem(STORAGE_KEY);
    else localStorage.setItem(STORAGE_KEY, name);
  }
}

/**
 * Resolve the effective env for a request: explicit URL param wins over the
 * persisted value. If neither is present, fall back to first available env.
 */
export function resolveEnv(urlParam: string | null, available: string[]): string | null {
  if (urlParam && available.includes(urlParam)) return urlParam;
  if (_env && available.includes(_env)) return _env;
  return available[0] ?? null;
}

<script lang="ts">
  import { createQuery } from '@tanstack/svelte-query';
  import { searchNodes } from '$lib/api/nodes';
  import type { NodeTarget } from '$lib/api/types';
  import NodeStatusBadge from './NodeStatusBadge.svelte';

  import { goto } from '$app/navigation';

  let { env, target = 'all' }: { env: string; target?: NodeTarget } = $props();

  let pageIndex = $state(0);
  const pageSize = 50;
  let searchInput = $state('');
  let debouncedSearch = $state('');
  let sortColumn = $state<string | undefined>(undefined);
  let sortDir = $state<'asc' | 'desc'>('asc');

  // Debounce the search input.
  let debounceTimer: ReturnType<typeof setTimeout> | undefined;
  $effect(() => {
    const v = searchInput;
    clearTimeout(debounceTimer);
    debounceTimer = setTimeout(() => { debouncedSearch = v; pageIndex = 0; }, 250);
    return () => clearTimeout(debounceTimer);
  });

  const query = $derived(createQuery({
    queryKey: ['nodes', env, target, pageIndex, pageSize, debouncedSearch, sortColumn, sortDir],
    queryFn: () => searchNodes(env, target, {
      page: pageIndex,
      pageSize,
      search: debouncedSearch,
      sortColumn,
      sortDir
    }),
    refetchInterval: 30_000,
    placeholderData: (prev) => prev   // keep showing previous page while next loads
  }));

  function toggleSort(col: string) {
    if (sortColumn === col) {
      sortDir = sortDir === 'asc' ? 'desc' : 'asc';
    } else {
      sortColumn = col;
      sortDir = 'asc';
    }
    // Reflect in URL so it's bookmarkable / back-button-friendly.
    const url = new URL(window.location.href);
    url.searchParams.set('sort', col);
    url.searchParams.set('dir', sortDir);
    goto(url.toString(), { replaceState: true, noScroll: true, keepFocus: true });
  }
</script>

    {/if}

    {/if}
  </div>

          {/each}
          <th class="text-left px-3 py-2 font-medium border-b">Status</th>
        </tr>

**Note on the cookie stub:** if osctrl-admin's session middleware verifies cookies cryptographically (most likely), the fake value above won't get past `handlerAuthCheck`. Two ways to make the e2e robust against this:

1. Intercept `/ui/api/bootstrap` and `/paginated-json/...` *before* the navigation and have Playwright skip the actual auth round-trip — the route stubs in the test above already do this. But Go still rejects the cookie at `handlerAuthCheck`, so you'd never reach `/ui/`. Workaround: also stub `/ui/` itself to return the built `index.html`. This makes the test essentially front-end-only.
2. Alternatively, run the test with auth disabled — osctrl-admin supports `AuthNone` (`flagParams.Service.Auth == config.AuthNone`). Add a CI-only config that runs the binary in `AuthNone` mode for the e2e.

Approach (2) is cleaner. Document the env-var or config-file flag your local osctrl uses to enable `AuthNone`, and run the binary with that flag in the e2e setup. **If you're not sure which flag, search:**

```bash
grep -rn "AuthNone\|auth.*none" cmd/admin/ pkg/config/ deploy/
```

Update the test's `test.beforeAll` (or a global setup file) to spawn the binary with that flag.
